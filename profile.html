function changeAvatar(){ let user = JSON.parse(localStorage.getItem("user")); let avatar = document.getElementById("avatarInput").value; if(!avatar){ alert("Введите ссылку"); return; } fetch("/profile/avatar",{ method:"POST", headers:{ "Content-Type":"application/json" }, body:JSON.stringify({ login:user.login, avatar:avatar }) }) .then(res=>res.json()) .then(data=>{ if(data.success){ user.avatar=avatar; localStorage.setItem( "user", JSON.stringify(user) ); document.getElementById("avatar").src=avatar; alert("Аватар изменён"); } }); }
